cmake_minimum_required(VERSION 3.20)
project(OpenNavicatControlledJavaScript C)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_LIBC OFF CACHE BOOL "" FORCE)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
if(NOT DEFINED QUICKJS_SOURCE_DIR)
  message(FATAL_ERROR "QUICKJS_SOURCE_DIR must be the verified official-source extraction")
endif()
add_subdirectory("${QUICKJS_SOURCE_DIR}" quickjs EXCLUDE_FROM_ALL)
add_executable(opennavicat-js runner.c)
target_link_libraries(opennavicat-js PRIVATE qjs)
if(MSVC)
  target_compile_definitions(opennavicat-js PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_options(opennavicat-js PRIVATE /STACK:16777216)
else()
  target_compile_options(opennavicat-js PRIVATE -Wall -Wextra -Werror)
endif()
